# Guanzhi 官制 - 勋官授予效果. root=受勋者; 纯 trait+旗标, 不建头衔
# 显示: 有地者经整合包风味值覆盖挂载, 无头衔者经 gzfv_titleless 风味条目挂载

# 授上柱国(视正二品)
guanzhi_xun_grant_01 = {
	guanzhi_clear_xun_effect = yes
	scope:recipient = {
		add_trait = gzx_01
		if = {
			limit = { is_landed = no }
			add_character_flag = guanzhi_xun_01
		}
		add_character_flag = guanzhi_xunholder
		guanzhi_fix_titleless_nickname_effect = yes
	}
	debug_log = "GUANZHI xun granted"
}
