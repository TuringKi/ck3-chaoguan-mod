# 自动授上柱国 (root=受勋者)
guanzhi_xun_auto_grant_01 = {
	if = {
		limit = {
			exists = liege
			# 勋阶跟人不跟地: 已有勋阶(任何来源)一律不动; 调整请手动授予
			NOT = { guanzhi_has_xun_trigger = yes }
			NOT = { any_held_title = { has_variable = guanzhi_is_xun } }
		}
		save_scope_as = recipient
		guanzhi_xun_grant_01 = yes
		add_character_flag = guanzhi_xun_auto
	}
}
