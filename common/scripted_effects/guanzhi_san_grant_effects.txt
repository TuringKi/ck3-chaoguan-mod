# Guanzhi 官制 - 散官授予效果. root=受官者; 无地分支在 scope:recipient 的 liege 作用域建头衔

# 授开府仪同三司(从一品)
guanzhi_san_grant_wen_01 = {
	guanzhi_clear_san_effect = yes
	# 散官只是品位凭证: 纯 trait+角色旗标, 不建头衔
	# 显示: 有地者经整合包风味值覆盖挂载(guanzhi_pack_mount_replace),
	#       无头衔者经昵称显示(风味化需命名头衔, 无头衔者不生效)
	scope:recipient = {
		add_trait = gzs_wen_01
		if = {
			limit = { is_landed = no }
			add_character_flag = guanzhi_san_wen_01
		}
		add_character_flag = guanzhi_sanholder
		# 无头衔者: 风味化需要命名头衔, 改用昵称显示(同整合包WJ将军号模式)
		guanzhi_fix_titleless_nickname_effect = yes
	}
	debug_log = "GUANZHI san granted to [recipient.GetShortUIName]"
}
