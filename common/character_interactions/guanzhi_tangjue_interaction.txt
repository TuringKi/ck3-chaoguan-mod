# Guanzhi 官制 - 唐代爵位互动 (9750 选爵级 -> 择地名; 9755 加实封)

guanzhi_grant_tangjue_interaction = {
	category = interaction_category_vassal
	desc = guanzhi_grant_tangjue_interaction_desc
	filter_tags = { guanzhi }

	is_shown = {
		scope:actor = {
			guanzhi_is_celestial_sovereign_trigger = yes
			has_character_flag = guanzhi_uses_tang_system
		}
		scope:recipient = {
			NOT = { this = scope:actor }
			is_adult = yes
			# 王公爵与侯爵不共存
			NOT = { has_trait_with_flag = guanzhi_wangjue }
			# 领地下所有人皆可授予: 廷臣(无liege链接, 单独判)/封臣/封臣之封臣及其廷臣
			OR = {
				is_courtier_of = scope:actor
				is_vassal_of = scope:actor
				liege = scope:actor
				top_liege = scope:actor
			}
		}
	}
	is_valid = {
		always = yes
	}

	auto_accept = yes
	on_accept = {
		scope:actor = {
			debug_log = "GUANZHI tangjue interaction accepted"
			trigger_event = { id = guanzhi.9750 }
		}
	}

	ai_will_do = {
		base = 0
	}
}
