# Guanzhi 官制 - 虚衔互动 (州牧/都督/大总管/检校; 9850 系统菜单)

guanzhi_grant_xuxian_interaction = {
	category = interaction_category_vassal
	desc = guanzhi_grant_xuxian_interaction_desc
	filter_tags = { guanzhi }

	is_shown = {
		scope:actor = {
			guanzhi_is_celestial_sovereign_trigger = yes
		}
		scope:recipient = {
			NOT = { this = scope:actor }
			is_adult = yes
			# 领地下所有人皆可授予: 廷臣(无liege链接, 单独判)/封臣/封臣之封臣及其廷臣
			OR = {
				is_courtier_of = scope:actor
				is_vassal_of = scope:actor
				liege = scope:actor
				top_liege = scope:actor
			}
			# 完全无头衔者不授(其显示走昵称通道会与整合包风味冲突双显);
			# 王公/朝官/使职等无地头衔持有者可授(有命名头衔, 风味链正常)
			exists = primary_title
		}
	}
	is_valid = {
		always = yes
	}

	auto_accept = yes
	on_accept = {
		scope:actor = {
			trigger_event = { id = guanzhi.9850 }
		}
	}

	ai_will_do = {
		base = 0
	}
}
