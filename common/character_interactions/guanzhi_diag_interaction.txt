# Guanzhi 官制 - 官爵状态诊断互动 (调试用, 用完可删)
# 右键任意人物使用, 状态写入 debug.log (搜 GZDIAG)

guanzhi_diag_rank_state_interaction = {
	category = interaction_category_vassal
	desc = guanzhi_diag_rank_state_interaction_desc
	filter_tags = { guanzhi }

	is_shown = {
		scope:actor = { always = yes }
		scope:recipient = { always = yes }
	}
	is_valid = {
		always = yes
	}

	auto_accept = yes
	on_accept = {
		scope:recipient = {
			debug_log = "GZDIAG ==== [recipient.GetShortUIName] ===="
			if = {
				limit = { is_landed = yes }
				debug_log = "GZDIAG landed=YES"
			}
			else = { debug_log = "GZDIAG landed=NO" }
			if = {
				limit = { guanzhi_has_san_trigger = yes }
				debug_log = "GZDIAG has_san_trigger=YES"
			}
			else = { debug_log = "GZDIAG has_san_trigger=NO" }
			if = {
				limit = { has_trait = gzs_wen_02 }
				debug_log = "GZDIAG trait wen_02=YES"
			}
			if = {
				limit = { has_trait = gzs_wen_01 }
				debug_log = "GZDIAG trait wen_01=YES"
			}
			if = {
				limit = { has_character_flag = guanzhi_san_wen_01 }
				debug_log = "GZDIAG flag wen_01=YES"
			}
			else = { debug_log = "GZDIAG flag wen_01=NO" }
			if = {
				limit = { has_character_flag = guanzhi_san_wen_02 }
				debug_log = "GZDIAG flag wen_02=YES"
			}
			else = { debug_log = "GZDIAG flag wen_02=NO" }
			if = {
				limit = { has_character_flag = guanzhi_sanholder }
				debug_log = "GZDIAG sanholder=YES"
			}
			else = { debug_log = "GZDIAG sanholder=NO" }
			if = {
				limit = { any_held_title = { has_variable = guanzhi_is_san } }
				debug_log = "GZDIAG holds_san_title=YES"
			}
			else = { debug_log = "GZDIAG holds_san_title=NO" }
			if = {
				limit = { has_nickname = nick_gz_rank_mount }
				debug_log = "GZDIAG nickname=YES"
			}
			else = { debug_log = "GZDIAG nickname=NO" }
			debug_log = "GZDIAG primary=[recipient.GetPrimaryTitle.GetNameNoTooltip]"
			debug_log = "GZDIAG san_display=[recipient.Custom('guanzhi_san_display')]"
			debug_log = "GZDIAG xun_display=[recipient.Custom('guanzhi_xun_display')]"
			debug_log = "GZDIAG jue_display=[recipient.Custom('guanzhi_jue_display')]"
			# ---- 虚衔/昵称状态 ----
			debug_log = "GZDIAG jj_display=[recipient.Custom('guanzhi_jj_display')]"
			debug_log = "GZDIAG zm_display=[recipient.Custom('guanzhi_zm_display')]"
			debug_log = "GZDIAG zgj_display=[recipient.Custom('guanzhi_zgj_display')]"
			debug_log = "GZDIAG jie_suffix=[recipient.Custom('guanzhi_jie_suffix')]"
			if = {
				limit = { has_any_nickname = yes }
				debug_log = "GZDIAG has_any_nickname=YES"
			}
			else = { debug_log = "GZDIAG has_any_nickname=NO" }
			if = { limit = { has_nickname = nick_gz_san } debug_log = "GZDIAG nick=san" }
			if = { limit = { has_nickname = nick_gz_xun } debug_log = "GZDIAG nick=xun" }
			if = { limit = { has_nickname = nick_gz_jj } debug_log = "GZDIAG nick=jj" }
			if = { limit = { has_nickname = nick_gz_tj } debug_log = "GZDIAG nick=tj" }
			if = { limit = { has_nickname = nick_gz_jue } debug_log = "GZDIAG nick=jue" }
			if = { limit = { has_nickname = nick_gz_zm } debug_log = "GZDIAG nick=zm" }
			if = { limit = { has_nickname = nick_gz_jie } debug_log = "GZDIAG nick=jie" }
			if = { limit = { has_nickname = nick_gz_zgj } debug_log = "GZDIAG nick=zgj" }
			if = { limit = { has_character_flag = guanzhi_zmholder } debug_log = "GZDIAG flag=zmholder" }
			if = { limit = { has_character_flag = guanzhi_jieholder } debug_log = "GZDIAG flag=jieholder" }
			if = { limit = { has_character_flag = guanzhi_zgjholder } debug_log = "GZDIAG flag=zgjholder" }
			if = { limit = { has_character_flag = guanzhi_jjholder } debug_log = "GZDIAG flag=jjholder" }
			# 双显排查: 父母为统治者时整合包 child 风味可能叠加昵称
			if = {
				limit = { any_parent = { is_ruler = yes } }
				debug_log = "GZDIAG parent_is_ruler=YES"
			}
			else = { debug_log = "GZDIAG parent_is_ruler=NO" }
			if = {
				limit = { has_character_flag = mytm_var_house_flavor_child }
				debug_log = "GZDIAG pack_child_flavor_flag=YES"
			}
			else = { debug_log = "GZDIAG pack_child_flavor_flag=NO" }
		}
	}

	ai_will_do = {
		base = 0
	}
}
