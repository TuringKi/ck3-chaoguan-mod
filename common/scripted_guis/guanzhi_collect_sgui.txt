# Guanzhi 官制 - 朝堂列表收集 (供 GUI dynamicgridbox 显示)

# 文官列表: 收集所有带朝官文官头衔的封臣
guanzhi_collect_civil_sgui = {
	scope = character
	effect = {
		clear_variable_list = guanzhi_officers_civil
		every_vassal = {
			limit = {
				any_held_title = { has_variable = guanzhi_office_civil }
			}
			root = {
				add_to_variable_list = {
					name = guanzhi_officers_civil
					target = prev
				}
			}
		}
	}
}
